```python
    """Show settings dialog."""
    try:
        from octomaster.gui.dialogs.settings_dialog import SettingsDialog

        dialog = SettingsDialog(self.config, self)
        if dialog.exec():
            # Сохраняем настройки
            new_config = dialog.get_config()
            self.config.update(new_config)

            # Сохраняем в файл
            self.save_config()

            logger.info("Settings saved successfully")

            # Показываем уведомление
            self.console_panel.append("✅ Settings saved")

            # Применяем новые настройки
            self.apply_config()

    except ImportError:
        logger.error("Settings dialog not found, creating it")
        # Создаем диалог если его нет
        self.create_settings_dialog()
    except Exception as e:
        logger.error(f"Failed to show settings: {e}")
        QMessageBox.critical(
            self, "Error",
```

```python
def export_workflow(self):
    """Export current workflow to Python or JSON."""
    if not self.current_workflow:
        QMessageBox.warning(
            self, "Export",
            "No workflow to export.\nCreate a workflow first."
        )
        return

    try:
        # Диалог выбора файла
        filename, filter_type = QFileDialog.getSaveFileName(
            self,
            "Export Workflow",
            f"{self.current_workflow.name}.py",
            "Python Files (*.py);;JSON Files (*.json);;All Files (*.*)"
        )

        if not filename:
            return  # Пользователь отменил

        # Определяем формат по расширению
        if filename.endswith('.json'):
            # Экспорт в JSON
            workflow_data = self.current_workflow.to_dict()
            with open(filename, 'w', encoding='utf-8') as f:
                json.dump(workflow_data, f, indent=2, ensure_ascii=False)
            logger.info(f"Workflow exported to JSON: {filename}")

        else:
            # Экспорт в Python
            python_code = self.current_workflow.to_python()
            with open(filename, 'w', encoding='utf-8') as f:
                f.write(python_code)
            logger.info(f"Workflow exported to Python: {filename}")

        # Показываем успех
        QMessageBox.information(
            self, "Export Successful",
            f"Workflow exported to:\n{filename}"
        )

        # Логируем в консоль
        self.console_panel.append(f"✅ Exported to {filename}")

    except PermissionError:
        QMessageBox.critical(
            self, "Export Failed",
            f"Permission denied:\n{filename}\n\nChoose different location."
        )
    except Exception as e:
        logger.error(f"Export failed: {e}", exc_info=True)
        QMessageBox.critical(
            self, "Export Failed",
            f"Failed to export workflow:\n{e}"
        )
```

```python
def create_parameter_field(self, param_name, param_value):
    """Create appropriate input field based on parameter type."""
    try:
        # Определяем тип параметра
        param_type = self.get_parameter_type(param_name, param_value)

        label = QLabel(param_name.replace('_', ' ').title() + ":")

        if param_type == 'bool':
            # Checkbox для boolean
            widget = QCheckBox()
            widget.setChecked(bool(param_value))
            widget.stateChanged.connect(
                lambda state: self.on_parameter_changed(param_name, bool(state))
            )

        elif param_type == 'choice':
            # ComboBox для выбора из списка
            widget = QComboBox()
            choices = self.get_parameter_choices(param_name)
            widget.addItems(choices)
            if param_value in choices:
                widget.setCurrentText(str(param_value))
            widget.currentTextChanged.connect(
                lambda value: self.on_parameter_changed(param_name, value)
            )

        elif param_type == 'multiline':
            # TextEdit для длинного текста
            widget = QTextEdit()
            widget.setPlainText(str(param_value) if param_value else "")
            widget.setMaximumHeight(100)
            widget.textChanged.connect(
                lambda: self.on_parameter_changed(
                    param_name,
                    widget.toPlainText()
                )
            )

        elif param_type == 'number':
            # SpinBox для чисел
            widget = QSpinBox()
            widget.setRange(0, 999999)
            widget.setValue(int(param_value) if param_value else 0)
            widget.valueChanged.connect(
                lambda value: self.on_parameter_changed(param_name, value)
            )

        else:
            # LineEdit для обычного текста (по умолчанию)
            widget = QLineEdit()
            widget.setText(str(param_value) if param_value else "")
            widget.textChanged.connect(
                lambda text: self.on_parameter_changed(param_name, text)
            )
            widget.setPlaceholderText(f"Enter {param_name}...")

        # Добавляем в layout
        self.params_layout.addRow(label, widget)

        # Сохраняем ссылку на виджет
        self.param_widgets[param_name] = widget

        logger.debug(f"Created {param_type} field for parameter '{param_name}'")

    except Exception as e:
        logger.error(f"Failed to create parameter field '{param_name}': {e}")
        # Fallback: простой текстовый ввод
        widget = QLineEdit(str(param_value) if param_value else "")
        self.params_layout.addRow(QLabel(param_name + ":"), widget)

def on_parameter_changed(self, param_name, new_value):
    """Handle parameter value change."""
    try:
        if not self.current_block:
            return

        # Обновляем значение в блоке
        self.current_block.set_parameter(param_name, new_value)

        # Логируем изменение
        logger.debug(f"Parameter '{param_name}' changed to: {new_value}")

        # Уведомляем об изменении
        if hasattr(self, 'parameter_changed'):
            self.parameter_changed.emit(param_name, new_value)

    except Exception as e:
        logger.error(f"Failed to update parameter '{param_name}': {e}")
```

```python
def initialize(self):
    """Initialize the plugin and load its resources."""
    try:
        logger.info(f"Initializing plugin: {self.name}")

        # Загружаем конфигурацию плагина
        self.load_config()

        # Инициализируем ресурсы
        self.load_resources()

        # Регистрируем блоки плагина
        self.register_blocks()

        # Подключаем обработчики событий
        self.connect_signals()

        self.initialized = True
        logger.info(f"Plugin '{self.name}' initialized successfully")

    except Exception as e:
        logger.error(f"Failed to initialize plugin '{self.name}': {e}")
        self.initialized = False
        raise

def shutdown(self):
    """Shutdown the plugin and cleanup resources."""
    try:
        if not self.initialized:
            return

        logger.info(f"Shutting down plugin: {self.name}")

        # Отключаем обработчики
        self.disconnect_signals()

        # Отменяем регистрацию блоков
        self.unregister_blocks()

        # Освобождаем ресурсы
        self.cleanup_resources()

        # Сохраняем состояние
        self.save_state()

        self.initialized = False
        logger.info(f"Plugin '{self.name}' shut down successfully")

    except Exception as e:
        logger.error(f"Error shutting down plugin '{self.name}': {e}")
```
